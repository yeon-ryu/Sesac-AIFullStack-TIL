### 연산자
**산술 연산자**
```javascript
// + - * / %
// 정수 / 정수 가 실수일 경우 실수로 반환된다

// 거듭제곱 연산자
a ** n // a 의 n 제곱 aⁿ
```

&nbsp;

**할당 연산자**
```javascript
c += 3; // c = c + 3;
c -= 2; // c = c - 2;
```

&nbsp;

**증감 연산자**
```javascript
let d = 5;
// 후위 증감 : 현재 값을 먼저 사용하고, 그 다음 값을 바꾼다.
console.log(d++); // 5
// 전위 증감 : 값을 먼저 바꾸고, 바뀐 값을 사용한다.
console.log(++d); // 7
console.log(--d); // 6
```

&nbsp;

**비교 연산자**
```javascript
// 좌항과 우항의 비교 연산자를 비교하고 그 결과를 boolean 값으로 반환
// if 나 for문과 같은 제어문의 조건식에서 주로 사용
// > < <= >=

/**
 * 동등 / 일치 비교 연산자
 * 동등 비교(==, !=) : 연산자는 먼저 암묵적 타입 변환을 통해 타입을 일치시킨 후 비교
 * 일치 비교(===, !==) : 연산자는 타입과 값이 모두 일치하는지 비교
 */
 
const x = 10;

// 1. 등등 연산자
console.log(x == 20); // false
console.log(x == '10'); // true

// 2. 일치 연산자 (일치 연산자를 주로 사용함)
console.log(x === 10); // true
console.log(x === '10'); // false

// 비교 연산자를 통해 문자열끼리 비교도 가능하며 유니코드 순으로 비교한다.
console.log('hello' === "hello"); // true
console.log('a' === 'A'); // false
```

**논리 연산자**
```javascript
a && b // AND
a || b // OR
! a // NOT

/**
 * 단축 평가
 * 표현식을 평가하는 도중 평가 결과가 확정 된 경우 나머지 평가 과정을 생략하는 것
 * 연산자의 결과가 확정나게한 피연산자를 그대로 반환
 */
 console.log('apple' || 'bannana'); // apple : Thruthy한 값인 apple 만 보고 true 로 확정하고 'apple' 이라고 확정시킨 애를 반환
 console.log(null || 'bannana'); // banana : banana 에서 확정했기에 'banana' 를 반환
 console.log('apple' && 'banana'); // banana : 전부 Truthy 한 값인지 확인
 console.log(0 && 1); // 0 : 0 에서 확정했기에 0 을 반환
 
 console.log(('apple' || 'banana') == true); // false : 비교 반환값이 'apple' 이기에 true 와 다르다
 console.log(('apple' || 'banana') === 'apple'); // true
 
 /** 
 * 삼항 연산자 : 조건식에 따라ㅣ 두 값 중 하나를 선택하는 연산자
 * 조건 ? 값 : 값2
 * 조건이 true 일 경우 값1, false 일 경우 값2 선택
 */
let age = 20;
let result = (age >= 20) ? "성인" : "미성년자";
console.log(result); // 성인
age = 15;
result = (age >= 20) ? "성인" : "미성년자";
console.log(result); // 미성년자
```

**비트 연산자**
```javascript
a & b // and
a | b // or
a ^ b // xor : 두 비트가 다를 경우 true
~a // not : 비트를 전부 반전
a << b // 비트를 왼쪽으로 이동시키고 오른쪽은 0을 채움 : a 를 왼쪽으로 b번 이동
a >> b // 부호를 유지하며 비트를 오른쪽으로 이동 : a 를 오른쪽으로 b번 이동
a >>> b // 비트를 오른쪽으로 이동시키고 왼쪽에 무조건 0을 채움 : a를 오른쪽으로 b번 이동
```

### 조건문, 반복문
**조건문**
```javascript
/**
 * 조건문: if-else
 * if(조건식) { 조건식이 true 인 경우 실행 구문 }
 * else { 조건식이 false 인 경우 실행 구문 }
 */
let temperature = 5;
if(temperature < 10) console.log("오늘은 추운 날씨입니다."); // 조건이 참일 때 실행
else console.log("오늘은 날씨가 괜찮습니다."); // 조건이 거짓일 때 실행

/**
 * else if : 여러 조건을 처리
 * if(조건식1) { 조건식1이 true 인 경우 실행 구문 }
 * else if(조건식2) { 조건식1이 false 이고 조건식 2가 true 인 경우 실행 구문 }
 * else { 모든 조건식이 false 인 경우 실행 구문 }
 */
let score = 75;
if(score >= 90) {
    console.log("성적이 매우 우수합니다.");
} else if(score >= 70) {
    console.log("성적이 보통입니다.");
} else if(score >= 50) {
    console.log("성적이 부족합니다.");
} else {
    console.log("공부가 많이 필요합니다.");
}

/**
 * switch문
 * 하나의 변수에 대해 여러 경우를 처리한다.
 * 비교 연산자는 일치 연산자(===)로 처리
 */
let num = 10;

switch(num) {
    case 0 : console.log("0"); break;
    case "10" : console.log("동등 연산자"); break;
    case 10 : console.log('일치 연산자'); break; // 여기서 멈춘다
    default : console.log('어쩌다 여기까지');
}
```

**반복문**
```javascript
/**
 * 반복문 for
 * 명확한 반복 횟수를 지정하여 코드 실행
 * 
 * for(초기화; 조건식; 증감식) { 반복 실행할 코드 }
 */

/**
 * 1. 초기화 : 반복 변수를 초기값으로 설정
 * 2. 조건식 : 반복을 계속할지 여부 결정. true 일 경우 반복이 실행
 * 3. 실행 구문
 * 4. 증감식 : 반복 변수를 조정하여 조건식에 영향을 줌
 */
for(let i = 1; i <= 5; i++) {
    console.log(`반복 횟수 : ${i}`);
}

/** 1부터 10까지 숫자 중 짝수만 출력 */
for(let i = 1; i <= 10; i++) {
    if(i % 2 === 0) console.log(i);
}

/**
 * while 반복문 : 조건이 참인 동안 코드를 반복 실행
 * for 문은 반복 횟수가 명확할 때, while 문은 반복을 멈출 조건이 더 중요할 때 사용하기 좋다.
 * while(조건) { 실행할 구문(증감식 포함) }
 */

let count = 3;

while(count > 0) {
    console.log(count);
    count--;
}

/**
 * do while 반복문
 * 최소 한 번 실행한 후 조건을 검사한다.
 * do { 실행할 구문 } while(조건)
 */
let number = 4;

do {
    console.log(number);
    number++;
} while(number <= 3);

/**
 * continue : 현재 반복문의 나머지 코드를 건너뛰고, 다음 반복으로 넘어간다.
 * break : 현재 반복문을 완전히 종료한다.
 */
for(let i = 1; i <= 10; i++) {
    if(i % 2 === 0) continue; // 이번 반복만 패스
    console.log(i);
}

for(let i = 1; i <= 10; i++) {
    if(i % 2 === 0) break; // 반복문 종료
    console.log(i);
}

/**
 * 중첩 반복문
 */
Loop1 : for(let i = 1; i <= 3; i++) {
    for(let j = 1; j <=3; j++) {
        if(j === 2) continue; // 가장 가까운 반복문 하나에만 영향을 준다.
        console.log(`i : ${i}, j : ${j}`);
        if(i > j) break Loop1;
    }
}
```

### 객체 (Object)
**Object**
```javascript
/**
 * object (객체)
 * 자바스크립트는 객체 기반 프로그래밍 언어로 원시 값을 제외한 나머지값(함수, 배열 정규 표현식 등)은 모두 객체이다.
 * 객체는 0개 이상의 프로퍼티로 구성된 집합이며 프로퍼티는 키(key)와 값(value)으로 구성된다.
 */

/** 객체 생성(객체 리터럴 방식) : const 변수명 = {}; */
const student = {
    // 키 : 값 쌍으로 구성된 프로퍼티
    /** 프로퍼티 : 객체의 상태를 나타내는 값 */
    name : "판다",
    age : 5,
    /** 메서드 : 객체 안에 들어가 있는 함수. 프로퍼티(상태 데이터)를 참조하고 조작할 수 있는 동작 */
    getInfo : function() { return `${this.name}은/는 ${this.age}세 입니다.` }
};

console.log(student.getInfo()); // 판다은/는 5세 입니다.
console.log(student); // { name: '판다', age: 5, getInfo: [Function: getInfo] }
```

**프로퍼티 접근**
```javascript
/**
 * 프로퍼티 접근
 * 1. 마침표 표기법
 * 2. 대괄호 표기법
 */

const dog = {
    name : '뽀삐',
    eat : function(food) {
        console.log(`${this.name}은 ${food}를 맛있게 먹어요.`);
    }
};

// 마침표 표기법
console.log(dog.name);
dog.eat('두부');

// 대괄호 표기법 (*프로퍼티 키를 문자열로 작성)
console.log(dog['name']);
console.log(dog'eat');

// 아래와 같은 키의 경우 반드시 대괄호 접근법으로 접근
const obj = {
    'dash-key' : 'dash-value',
    0 : 1
}

console.log(obj[`0`]);
console.log(obj[`dash-key`]);

// key 가 변수에 담겨 있을 때
const prop = 'name';
console.log(dog.prop);
console.log(dog[prop]); // dog['name'] 과 동일
```

**프로퍼티 수정/추가/삭제**
```javascript
/**
 * const 객체의 프로퍼티 내부 값을 변경, 추가, 삭제 하는 것은 가능하지만 객체 자체를 재할당 하는 것은 불가능하다.
 */
const dog = {
    name : '뽀삐'
}

// 이미 존재하는 프로퍼티에 값을 할당하면 프로퍼티 값 갱신
dog.name = '감자'; // dog['name'] = 감자;

// 존재하지 않는 프로퍼티에 값을 할당하면 프로퍼티가 동적으로 생성된다.
dog.age = 5;
console.log(dog); // { name: '감자', age: 5 }

// 삭제 : delete 연산자 사용하여 프로퍼티를 삭제한다.
delete dog.age;
console.log(dog); // { name: '감자' }

// 객체 통째로 재할당 불가 (TypeError: Assignment to constant variable.)
// dog = { name : '초코' };

// 객체에서 특정 프로퍼티 존재 확인(boolean) : in 연산자
console.log('name' in dog); // true
console.log('age' in dog); // false

// 객체 순회 (for...in) : 객체의 프로퍼티 키를 하나씩 꺼내는 반복문
for(const key in dog) {
    console.log(`키 : ${key} 값 : ${dog[key]}`);
}
```
