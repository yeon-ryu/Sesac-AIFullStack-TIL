const fruits = ['사과', '바나나', '포도', '오렌지'];

            ))}    
        </ul>

## Hook
### useEffect
useEffect : 화면이 그려진 뒤 실행할 부수 효과(side effect)를 작성하는 자리  
`useEffect(() => { 실행 내용 });`  
콘솔 출력, 타이머 등록, 이벤트 등록, API 요청 등
```html
<script type="text/babel">
   
    const { useEffect } = React;

    function MessagePrinter() {
        console.log('렌더링'); // 1

        // 화면을 그리는 코드가 아니라, 
        // 화면이 그려진 뒤 실행할 부수 효과(side effect)를 작성하는 자리
        // 콘솔 출력, 타이머 등록, 이벤트 등록, API 요청 등
        useEffect(() => {
            console.log('렌더링 이후 동작...'); // 3
            console.log(message);
        });

        return (
            <h1>{console.log('렌더링 시 출력') /* 2 */}{message}</h1>
        )
    }

    const message = '안녕하세요!' // 4

    function App() {
            return (
                <>
                    <MessagePrinter/>
                </>
            )
    }
   
   ReactDOM.createRoot(document.getElementById('root')).render(<App/>);
</script>
```

#### mount
Mount : useEffect 훅의 의존성 배열을 빈 배열로 하면 컴포넌트가 처음 화면에 나타날 때(Mount) 단 한번만 실행됨  
`useEffect(() => { 실행 내용 }, []);`  
```jsx
const { useState, useEffect } = React;

function TimePrinter() {

    const [time, setTime] = useState(new Date().toLocaleTimeString());

    console.log('렌더링..')

    useEffect(
        () => { console.log('useEffect 동작...')},  // 수행할 동작
        []);    // 의존성 배열을 빈 배열로 설정하면 업데이트 시점에는 동작하지 않는다.
        // 렌더링.. useEffect 동작... 이 출력된 후 현재 시간 버튼을 눌러 setTime 으로 업데이트 되더라도 렌더링... 만 출력됨

    return (
        <>
            <button onClick={() => setTime(new Date().toLocaleTimeString())}>현재시간</button>
            <h1>{time}</h1>
        </>
    )
}
```

#### lifecycle
- **Rendering** : 의존성 배열 생략 : 매 렌더링마다 실행  
    `useEffect(() => { 실행 내용 });`
- **Mount** : 빈 의존성 배열 [] : Mount(첫 렌더링) 시에만 실행. 주로 초기 데이터 로딩(API 호출) 등에 사용  
    `useEffect(() => { 실행 내용 }, []);` 
- **Update** : 의존성 배열에 특정 값 지정 : 해당 값이 변경될 때만 실행  
    `useEffect(() => { 실행 내용 }, [state 상태값]);`
- **Unmount** : Cleanup 함수: useEffect가 반환하는 함수. (useEffect 의 실행 함수 안 return 함수)  
    컴포넌트가 화면에서 사라지기 직전(Unmount)에 호출된다.  
    의존성을 넣으면 의존성 값이 update 될 때마다 Unmount 하고 재실행 한다.  
    `useEffect(() => { 실행 내용; return () => { 사라지기 직전 실행 } });`
```html
<div id="root"></div>

<script type="text/babel">

    const {useState, useEffect} = React;

    console.log(React);
    
    function Counter() {

        const [count, setCount] = useState(0);
        const [otherCount, setOtherCount] = useState(0);

        // 일단 첫 마운트 시점에는 무조건 한번 동작

        // 1. 의존성 배열 생략 : 매 렌더링마다 실행
        useEffect(() => {
            console.log('렌더링될 때마다 실행')
        })

        // 2. 빈 의존성 배열 [] : Mount(첫 렌더링) 시에만 실행
        // 주로 초기 데이터 로딩(API 호출) 등에 사용
        useEffect(() => {
            console.log('컴포넌트가 처음 화면에 나타남(Mount)');
        }, [])

        // 3. 의존성 배열에 특정 값 지정 : 해당 값이 변경될 때만 실행
        useEffect(() => {
            console.log(`count값이 ${count}로 변경되어, 이 useEffect가 실행`);
        }, [count])

        console.log('Counter 컴포넌트 렌더링');

        return (
            <>
                <h2>단순 카운터</h2>
                <p>{count}번 클릭 했습니다.</p>
                <button onClick={() => setCount(count + 1)}>+1</button>
                <p>other count: {otherCount}</p>
                <button onClick={() => setOtherCount(otherCount + 1)}>+1</button>
            </>
        )
    }

    function Timer() {

        const [test, setTest] = useState(0);

        useEffect(() => {
            console.log('타이머가 시작되었습니다(Mount)');

            const timerId = setInterval(() => {
                console.log('...1초 경과...');
                // setTest(test + 1);
            }, 1000);

            // Cleanup 함수: useEffect가 반환하는 함수
            // 컴포넌트가 화면에서 사라지기 직전(Unmount)에 호출된다
            // 의존성이 바뀌어 재실행하는 경우에는 이전 작업을 정리하고 새 값으로 작업을 시작한다.
            return () => {
                console.log('타이머를 정리합니다(Unmount)');
                clearInterval(timerId); // 생성했던 타이머 제거
            }
        }, [test]); // 이 안에 의존성을 넣어주면 의존성이 Update 될 때마다 Unmount 동작한다.

        return (
            <h3>타이머가 동작 중입니다.(콘솔을 확인하세요)</h3>
        )
    }
   
    function App() {
            const [showTimer, setShowTimer] = useState(true);

            return (
                <>
                    <Counter/>
                    <hr/>
                    <button onClick={() => setShowTimer(!showTimer)}>타이머 보이기/숨기기</button>
                    {showTimer && <Timer/>}
                </>
            )
    }
   
   ReactDOM.createRoot(document.getElementById('root')).render(<App/>);
</script>
```
