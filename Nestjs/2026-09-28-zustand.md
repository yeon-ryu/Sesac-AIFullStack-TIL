# 전역 상태 관리 (Zustand)
store : 전역 상태를 저장하는 곳  
Zustand : 전역 상태 관리 라이브러리  

## 전역 상태 관리가 필요한 이유
- props drilling : 중첩된 구조에서 props 가 하위 컴포넌트로 가기 위해 중간에 사용하지도 않으면서 전달만 하는 현상
- 해당 현상을 방지하기 위해 어디서든 같은 상태를 공유하는 전역 상태(Global State)를 사용

## Zustand 설치 및 사용
```bash
npm install zustand
```
- set : 상태를 바꾸는 스위치 (zustand store 에 저장된 상태를 변경하는 함수)
- state : 현재 store 에 있는 데이터의 '스냅샷'

1. zutand 가 제공하는 create 함수를 사용하여 store 생성 (store 은 훅처럼 사용해야 하기에 use 로 시작)
```javascript
// src/store/useStore.js
import { create } from "zustand";

// set : 상태를 바꾸는 스위치
// state : 현재 stroe 에 있는 데이터의 '스냅샷'
export const useStore = create((set) => ({
    count : 0,  // 데이터
    text : "",
    user : {},

    increase : () => set((state) => ({count : state.count + 1})),    // 데이터를 조작하는 방법
    decrease : () => set((state) => ({count : state.count - 1})),

    // 스프레드 연산자를 사용하지 않더라도 set 이 자동으로 얕은 병합을 한다. 다만 중첩된 객체의 경우 스프레드 연산자가 필요
    setText : (value) => set({text : value}),
    setAge : (value) => set((state) => ({user : {...state.user, age : value}}))
}));
```

2. store 에서 필요한 값 가져와서 사용
```javascript
// src/app/page.js
'use client';

import Child from "./components/Child";
import { useStore } from "@/store/useStore";

export default function Home() {
  const { count } = useStore();

  return (
    <>
      <h1>Props Drilling</h1>
      <h2>Page 에서 보는 count : {count}</h2>
      <Child />
    </>
  );
}
```
```javascript
// src/app/component/GrandChild.js
'use client';

import { useStore } from "@/store/useStore";

export default function GrandChild() {

    const { count, increase, text, setText } = useStore();

    return <>
        <h2>GrandChild count : {count}</h2>
        <button onClick={increase}>+1</button>

        <input type="text" onChange={(e) => setText(e.target.value)} />
        <p>{text}</p>
    </>
}
```

## Zustand 심화
- 배열은 불변성 때문에 기존 배열을 직접 수정하면 안된다.
    ```javascript
    addItem : (item) => set(state => ({items : [...state.items, item]})),
    ```
### 비동기 패턴
비동기 통신 로직 자체를 store 안에 넣을 수 있다.
```javascript
// src/store/useUserStore.js
import { create } from "zustand";

export const useUserStore = create((set) => ({
    user : null,
    loading : false,

    login : (user) => set({user : user}),
    logout : () => set({user : null}),

    // 비동기 액션
    fetchUser : async() => {
        set({ loading : true }); // 통신 시작 시 로딩 true

        try {
            const res = await fetch('https://jsonplaceholder.typicode.com/users/1');
            if(!res.ok) throw new Error('');
            const data = await res.json();

            set({user : data});
        } catch (e) {
            console.error('데이터 가져오기 실패', e);
        } finally {
            set({ loading : false });
        }
    }
}));
```
```javascript
// app/async/page.js
'use client'

import { useUserStore } from "@/store/useUserStore";
import { useEffect } from "react";

export default function AsyncPage() {
    const { user, loading, fetchUser } = useUserStore();

    useEffect(() => {
        fetchUser();
    }, [fetchUser]);

    if(loading) return<h1>데이터 로딩 중...</h1>

    return <>
        <h1>서버 데이터와 Zustasnd 연동 (비동기)</h1>
        {user ? (
            <div>
                <h3>이름 : {user.name}</h3>
                <p>이메일 : {user.email}</p>
            </div>
        ) : (
            <p>유저 정보가 없습니다.</p>
        )}

        <button onClick={fetchUser}>데이터 다시 불러오기</button>
    </>
}
```

### Persist (localStorage)
기존의 zustand 는 메모리에 존재하기에 새로고침하면 초기값으로 돌아간다.  
persist 는 변경 사항을 localStorage 에 저장하여 새로고침하더라도 상태를 유지하게 해주는 미들웨어이다.  
상태값(ex. theme)만 저장되고 함수(ex. toggleTheme)는 저장되지 않는다.  
`persist(store 메소드, 설정 객체)`
```javascript
// src/store/usePersistStore.js
import { create } from "zustand";
import { persist } from "zustand/middleware";

export const usePersistStore = create(
    persist(
        // store 내용
        set => ({
            theme : "light",
            toggleTheme : () => set(state => ({theme : state.theme === "light" ? "dark" : "light"}))
        }),

        // 저장 설정 (localStorage 에 저장할 때 사용할 키 이름) : 개발자 도구에서 확인 가능
        {name : "theme-storage"}
    )
)
```

## Hydration 에러
서버(기본값)과 브라우저가 각자 다른 값을 그리려 할 때 발생할 수 있다.  
그러므로 렌더링 후에 그리게 만드느 것이 오류를 방지하는데 유용하다.  
1. 서버에서 HTML 생성 : 서버는 localStorage 값을 읽을 수 없기에 초기값을 기준으로 HTML 을 만든다.
2. 브라우저가 HTML 표시 : 브라우저가 저장한 값으로 만들려고 한다.
3. React 가 브라우저 실행
4. 기존 HTML 에 이벤트, 상태 연결 : Hydration (서버와 브라우저가 그릴려는 HTML 이 같은지 확인)
```javascript
'use client'
import { usePersistStore } from "@/store/usePersistStore";
import { useEffect, useState } from "react";

export default function PersistPage() {
    const { theme, toggleTheme } = usePersistStore();

    // Hydration 에러 해결 - 서버와 브라우저의 첫 렌더링 결과를 맞추기 위한 상태
    const [isMounted, setIsMounted] = useState(false);

    useEffect(() => {
        setIsMounted(true);
    }, [])

    if(!isMounted) return <div>로딩 중...</div>

    return  (
        <div style = {{background : theme === "dark" ? "#333" : "#fff", color : theme === "dark" ? "white" : "black"}}>
            <p>현재 테마 : {theme}</p>
            <button onClick={toggleTheme}>테마 변경</button>
        </div>
    )
}
```
skipHydration : 수동적으로 복원이 필요할 경우 하이드레이션 에러가 발생하지 않도록 하는 기능을 zukstand 에서 제공해주고 있다.

# Json Server
json 이라는 파일을 읽어서 GET, POST, PUT, PATCH, DELETE 를 사용할 수 있도록 Mock API 를 제작해주는 라이브러리

## JSON 서버 설치
```bash
npm install -D json-server
```
`-D` : 개발/실습에 사용하는 도구 (Dev dependency) 란 설정  

1. json 서버 포트 지정
    db.json 을 읽어서 4000 번 포트로 json-server 실행
    ```jsx
    // package.json
    // npm run server
      "scripts": {
        "dev": "next dev",
        "build": "next build",
        "start": "next start",
        "lint": "eslint",
        "server": "json-server db.json --port 4000"
      }
    ```

2. 프로젝트 최상위에 db.json 파일 작성
    ```jsx
    {
        "posts" : [
            {
                "id" : "1",
                "title" : "첫 번째 게시글",
                "author" : "판다"
            },
            {
                "id" : "2",
                "title" : "HTTP 요청 연습",
                "author" : "코알라"
            }
        ]
    }
    ```
    JSON : JavaScript Object Notation (자바스크립트 객체 표기법)  
    `/posts` 라는 API 경로를 자동으로 만든다.

3. 각자 다른 터미널을 열어서 npm run dev, npm run server 로 실행

## HTTP Method
GET : Read `/posts, /posts/1`  
POST : Create `/posts`  
PUT : Update `/posts/1`  
PATCH : Update `/posts/1`  
DELETE : Delete `/posts/1`  

## Mock API
```javascript
// src/api/postAPI.js
const BASE_URL = "http://localhost:4000/posts";

export const postApi = {
    getPosts : async() => {
        const response = await fetch(BASE_URL);
        if(!response.ok) throw new Error('게시글 목록을 가져오지 못했습니다.');
        return response.json();
    },

    getPost : async(id) => {
        const response = await fetch(BASE_URL + "/" + id);
        if(!response.ok) throw new Error('게시글을 가져오지 못했습니다.');
        return response.json();
    },

    createPost : async(newPost) => {
        const response = await fetch(BASE_URL, {
            method : "POST",
            headers : { "Content-Type" : "application/json" },
            // stringify : JS 객체를 JSON 본문으로 보낼 때 사용
            body : JSON.stringify(newPost)
        });
        if(!response.ok) throw new Error('게시글 등록에 실패했습니다.');
        
        return response.json(); // id 는 자동으로 생성되고 이러면 추가된 값이 반환된다.
    },

    updatePost : async(id, updatedTitle) => {
        const response = await fetch(`${BASE_URL}/${id}`, {
            method : "PATCH",
            headers : {"Content-Typle" : "application/json"},
            body : JSON.stringify({title : updatedTitle})
        });
        if(!response.ok) throw new Error('게시글 수정에 실패했습니다.');

        return response.json();
    },

    deletePost : async(id) => {
        const response = await fetch(`${BASE_URL}/${id}`, {
            method : "DELETE"
        });
        if(!response.ok) throw new Error('게시글 삭제에 실패했습니다.');

        return response.ok; // 성공 여부만 반환
    }
}
```
사용 예시
```javascript
// src/app/crud-test/page.js
'use client'

import { postApi } from "@/api/postAPI";
import { useEffect, useState } from "react";

export default function CRUDPage() {
    const [posts, setPosts] = useState([]);
    const [loading, setLoading] = useState(false);
    const [error, setError] = useState(null);

    const loadPosts = async() => {
        try {
            setLoading(true);
            setError(null);

            const data = await postApi.getPosts();
            setPosts(data);
        } catch(e) {
            setError(e.message);
        } finally {
            setLoading(false);
        }
    }

    const handleAdd = async() => {
        try {
            setError(null);

            await postApi.createPost({
                title : "새로운 게시글 " + Date.now(),
                author : "익명"
            });

            await loadPosts();
        } catch(e) {
            setError(e.message);
        }
    }

    const handleUpdate = async(id) => {
        try {
            setError(null);

            await postApi.updatePost(id, "수정한 타이틀");
            await loadPosts();

        } catch(e) {
            setError(e.message);
        }
    }

    const handleDelete = async(id) => {
        const shoudDelete = window.confirm('정말 삭제할까요?'); // 확인 창 (boolean 값 반환)

        if(!shoudDelete) return;

        try {
            setError(null);

            await postApi.deletePost(id);
            await loadPosts();

        } catch(e) {
            setError(e.message);
        }
    }

    useEffect(() => {
        loadPosts();
    }, []);

    if(loading) return <p>게시글 불러오는 중...</p>;
    if(error) return <p>오류 : {error}</p>;

    return <>
        <h1>게시글 관리</h1>
        {posts.map(post => <div key={post.id} style={{border : "1px solid black", padding : "5px", margin : "5px"}}>
            <h3>{post.title}</h3>
            <p>작성자 : {post.author}</p>
            <button onClick={() => handleUpdate(post.id)}>제목 수정</button>
            <button onClick={() => handleDelete(post.id)}>게시글 삭제</button>
        </div>)}
        <button onClick={handleAdd}>새 글 등록</button>
    </>
}
```

# TanStack Query
요청 로딩, 성공 여부, 에러 등 상태를 관리해주는 라이브러리  
queryKey 값으로 요청 결과를 캐시에 저장해 놓고 찾을 때도 키 값으로 찾아온다. 다른 컴포넌트가 같은 쿼리를 사용하면 캐시되어 있는 값을 활용한다. → 기존 데이터를 빠르게 표시해준다.

## 설치
```bash
npm install tanstack/react-query
```

## 사전 설정
```jsx
// src/app/Providers.js
'use client'

import { QueryClient, QueryClientProvider } from "@tanstack/react-query";
import { useState } from "react";

export default function Providers({ children }) {
    // QueryClient : Query 데이터와 캐시를 관리하는 객체
    // queryClient 에 QueryClient 객체를 할당
    const [queryClient] = useState(() => new QueryClient());

    // QueryClientProvider : QueryClient 를 하위 컴포넌트에 공급하는 컴포넌트
    return (
        <QueryClientProvider client={queryClient}>
            {children}
        </QueryClientProvider>
    )
}
```
```jsx
// src/app/layout.js
export default function RootLayout({ children }) {
  return (
    <html lang="en" className={`${geistSans.variable} ${geistMono.variable}`}>
      <body><Providers>{children}</Providers></body>
    </html>
  );
}
```

## 사용
```jsx
// src/api/userApi.js
export const fetchUsers = async() => {
    const response = await fetch('https://jsonplaceholder.typicode.com/users');

    if(!response.ok) throw new Error('데이터를 가져오는데 실패!');

    return response.json();
}
```
```jsx
// src/app/query/page.js
'use client'

import { fetchUsers } from "@/api/userApi";
import { useQuery } from "@tanstack/react-query";

export default function QueryPage() {
    const query = useQuery({

         // 서버 데이터를 구분하는 고유한 이름 (요청 결과를 캐시에 저장해 놓고 찾을 때도 키 값으로 찾아온다.)
        queryKey : ["users"], // 사용자 전체 목록
        // queryKey : ["users", 1], // 1번 사용자 보기\

        // query function
        queryFn : fetchUsers // 실제로 데이터를 가져오는 비동기 함수
    });

    // status, fetchStatus, data, error, isLoading 같은 상태가 들어있다.
    console.log(query);
    
    // 성공한 결과는 data 에, 실패한 결과는 error 에 들어간다. 객체 구조 분해 할당으로 꺼내서 바로 사용 가능
    const {data, isLoading, isError, error, refetch, isFetching} = query;
    // refetch : 이 쿼리를 다시 요청하고 싶을 때 사용
    // isLoading : 처음으로 데이터를 불러올 때, isFetching : 서버에 요청 진행 중인가 (이미 캐시되어 있는 데이터 재요청도 포함)
    
    if(isLoading) {
        return <h1>데이터를 불러오는 중...</h1>
    }
    if(isError) {
        return <>
            <h1>사용자 정보를 가져오지 못함</h1>
            <p>{error.message}</p>
            <button onClick={refetch}>다시 시도</button>
        </>
    }

    return <>
        <h1>사용자 목록</h1>
        {data.map((user) => (<div key={user.id}>
            <strong>{user.name}</strong>
            <p>{user.email}</p>
        </div>))}
    </>
}
```

## Lazy
컴포넌트를 초기에 렌더링 하지 않고 실제로 사용할 때 렌더링 하게 해주는 함수.  
클라이언트에서 동적으로 컴포넌트를 불러오는 방식이기에, 서버 컴포넌트의 경우 서버에서 이미 전부 렌더링 되었기에 소용없다.  
lazy 를 사용한 컴포넌트는 Suspense 로 감싸는게 필수이다.
```jsx
import { lazy, Suspense } from 'react';

// 1. lazy를 이용해 동적 import 지정 (default export 필수)
const MyComponent = lazy(() => import('./MyComponent'));

function App() {
  return (
    // 2. Suspense로 감싸고 로딩 중 보여줄 UI 설정
    <Suspense fallback={<div>로딩 중...</div>}>
      <MyComponent />
    </Suspense>
  );
}
```

lazy 는 default export 된 모듈만 지원한다.  
프로젝트에 적용한다면 기존 useSearchParams 를 위해 Suspense 했던걸 빼내고 기존 컴포넌트를 export default 로 다시 바꾸고, 탭 실행 페이지 화면(헤더와 같은 레벨?의 상위 화면) 에서 탭에 있는 화면 컴포넌트를 그릴 때 Suspense 태그로 감싸는걸로 바꾸는게 좋을 것 같다.
```jsx
import { Suspense } from "react";
import { useTabStore } from "./useTabStore";
import { pageRegistry } from "./registry";

export function TabContent() {
  const tabs = useTabStore((s) => s.tabs);
  const activeId = useTabStore((s) => s.activeId);

  return (
    <Suspense fallback={<div>로딩중...</div>}>
      {tabs.map((tab) => {
        const Page = pageRegistry[tab.pageKey].component;
        return (
          // 비활성 탭도 언마운트하지 않고 숨김 → 입력값, 스크롤, 로컬 state 유지
          <div key={tab.id} style={{ display: tab.id === activeId ? "block" : "none" }}>
            <Page {...tab.params} />
          </div>
        );
      })}
    </Suspense>
  );
}
```
