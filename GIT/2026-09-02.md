## 로컬 저장소 사용
```bash
# git 저장소로 지정
git init

# 현재 상태 확인
git status

# 로그 확인
git log --oneline

# add -> staging area
git add 파일명
# 모든 파일 올리기
git add .

# commit -> local repository
git commit -m '커밋 메세지'

# commit 되돌리기 (되돌리는 커밋 생성)
git revert 목표커밋ID
# 편집기 화면 INSERT 모드 커밋 메세지 -> esc -> :wq (저장 후 종료)
```
&nbsp;

## 원격 저장소 연결
```bash
# 로컬 브랜치가 master 일 경우 마스터 브랜치 이름을 main 으로 변경
git branch -M main

# 로컬과 원격 저장소 연결 (해당 원격 프로젝트를 origin 이란 이름으로 연결)
git remote add origin git원격https

# 원격 연결 확인
git remote -v
```
&nbsp;

## 원격 저장소 통신
```bash
# push -> remote repository (-u : git 이 어느 원격 브랜치와 연결되는지 기억)
git push -u origin(원격저장소명) 원격브랜치명
git push 원격저장소명 원격브랜치명(새 원격 브랜치 작성도 가능)
git push

# remote repository 변경 사항 확인
git fetch

# remote repository -> local
git pull origin(원격저장소명) 원격브랜치명
git pull

# 아예 새로 프로젝트 통째로 복제
git clone git원격https
```
&nbsp;

## 브랜치

```bash
# 브랜치 생성
git branch 브랜치명

# 브랜치 확인
git branch

# 브랜치 변경
git switch 브랜치명
git checkout 브랜치명

# 브랜치 생성과 동시에 이동
git switch -c 브랜치명
git checkout -b 브랜치명

# merge
git merge 병합가져올브랜치명
# merge conflict 발생 시 직접 코드에서 수정하고 add commit : (main|MERGING)
# <<<<<<< HEAD : 현재 브랜치 내용, >>>>>>> feature/conflict 상대 브랜치 내용
```
&nbsp;

## 스태시
```bash
# 임시 파일 보관
git stash

# 스태시 확인
git stash list

# 스태시 적용하기
git stash apply

# 스태시 제거
git stash drop
```
